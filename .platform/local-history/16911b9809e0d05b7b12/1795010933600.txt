name: Release APK

on:
  push:
    tags:
      - "v*"
  workflow_dispatch:

permissions:
  contents: write

jobs:
  release:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Set up JDK 17
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: "17"

      # The repo has no Gradle wrapper, so install Gradle 8.10 and call it directly.
      - name: Set up Gradle
        uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: 8.10

      - name: Compute version from tag
        run: |
          if [ "$GITHUB_REF_TYPE" = "tag" ]; then
            V="${GITHUB_REF_NAME#v}"
          else
            V="1.0.$(date +%Y%m%d%H%M)"
          fi
          echo "VERSION_NAME=$V" >> "$GITHUB_ENV"
          echo "VERSION_CODE=$(( $(date +%s) / 60 ))" >> "$GITHUB_ENV"

      - name: Decode signing keystore
        env:
          KEYSTORE_BASE64: ${{ secrets.KEYSTORE_BASE64 }}
        run: |
          if [ -n "$KEYSTORE_BASE64" ]; then
            echo "$KEYSTORE_BASE64" | base64 -d > "$GITHUB_WORKSPACE/keysbook/release.keystore"
            echo "KEYSTORE_FILE=$GITHUB_WORKSPACE/keysbook/release.keystore" >> "$GITHUB_ENV"
          else
            echo "No keystore secret set - release APK will be debug-signed."
          fi

      - name: Build debug and release APKs
        working-directory: keysbook
        env:
          KEYSTORE_PASSWORD: ${{ secrets.KEYSTORE_PASSWORD }}
          KEY_ALIAS: ${{ secrets.KEY_ALIAS }}
          KEY_PASSWORD: ${{ secrets.KEY_PASSWORD }}
        run: gradle assembleDebug assembleRelease --stacktrace

      - name: Collect artifacts
        run: |
          mkdir -p dist
          cp keysbook/build/outputs/apk/debug/*.apk dist/keysbook-debug.apk
          cp keysbook/build/outputs/apk/release/*.apk dist/keysbook-release.apk
          cd dist
          sha256sum *.apk > SHA256SUMS.txt
          ls -la

      - name: Upload artifacts
        uses: actions/upload-artifact@v4
        with:
          name: keysbook-apks
          path: dist/*

      - name: Create GitHub release
        if: github.ref_type == 'tag'
        env:
          GH_TOKEN: ${{ github.token }}
        run: |
          TAG="$GITHUB_REF_NAME"
          NOTES="$(mktemp)"
          {
            echo "## KeysBook $TAG"
            echo
            echo "- **keysbook-debug.apk** - debug build"
            echo "- **keysbook-release.apk** - release build"
            echo "- Verify: \`sha256sum -c SHA256SUMS.txt\`"
          } > "$NOTES"
          if gh release view "$TAG" >/dev/null 2>&1; then
            gh release upload "$TAG" dist/* --clobber
          else
            gh release create "$TAG" dist/* --title "KeysBook $TAG" --notes-file "$NOTES"
          fi
